function salarySourceKey(x){return x.recurrenceId?`rule:${x.recurrenceId}`:JSON.stringify([x.name,x.account])}
function salaryMatches(x){const source=st.settings.salarySourceDetails;return x.type===`income`&&(salarySourceKey(x)===st.settings.salarySource||source&&(source.recurrenceId&&source.recurrenceId===x.recurrenceId||source.name===x.name&&source.account===x.account))}
function nextMonthDay(date){const d=new Date(`${date}T12:00:00`),month=new Date(d.getFullYear(),d.getMonth()+1,1);return inMonthDate(`${month.getFullYear()}-${String(month.getMonth()+1).padStart(2,`0`)}`,date)}
function dayBefore(date){const d=new Date(`${date}T12:00:00`);d.setDate(d.getDate()-1);return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,`0`)}-${String(d.getDate()).padStart(2,`0`)}`}
function prettyFinancialDate(date){return new Intl.DateTimeFormat(`ar-EG`,{day:`numeric`,month:`long`,year:`numeric`}).format(new Date(`${date}T12:00:00`))}
function financialPeriod(){const today=localToday(),view=st.settings.financialView===`month`?`month`:`cycle`,month=today.slice(0,7),rows=all();if(view===`month`)return {view,start:`${month}-01`,asOf:today,until:nextMonthDay(`${month}-01`),title:`هذا الشهر`,ready:true,estimated:false};
const salaries=rows.filter(x=>salaryMatches(x)&&x.completed===true&&validDate(x.date)&&x.date<=today).sort((a,b)=>b.date.localeCompare(a.date));const last=salaries[0];if(!last)return {view,start:today,asOf:today,until:today,title:`من آخر راتب`,ready:false,estimated:false};
const scheduled=rows.filter(x=>salaryMatches(x)&&validDate(x.date)&&x.date>today).sort((a,b)=>a.date.localeCompare(b.date))[0];let until=scheduled?.date,estimated=!until;
if(!until){const rule=st.recurrences?.find(r=>r.enabled&&salaryMatches({...r.template,recurrenceId:r.id}));let candidate=nextMonthDay(last.date);if(rule)candidate=recurrenceDate(rule,candidate.slice(0,7));while(candidate<=today){candidate=nextMonthDay(candidate);if(rule)candidate=recurrenceDate(rule,candidate.slice(0,7))}until=candidate}
return {view,start:last.date,asOf:today,until,title:`من آخر راتب`,ready:true,estimated,salary:last};}
function financialSummary(period=financialPeriod()){const actual=all().filter(x=>x.completed===true&&validDate(x.date)&&x.date>=period.start&&x.date<=period.asOf);const income=roundMoney(actual.filter(x=>x.type===`income`).reduce((s,x)=>s+n(x.amount),0)),expense=roundMoney(actual.filter(x=>x.type===`expense`).reduce((s,x)=>s+n(x.amount),0));const pending=obligationEntries().filter(({item:x})=>x.type===`expense`&&!x.obligationId&&x.completed!==true&&(!validDate(x.date)||x.date<period.until)).map(({month,item})=>({month,item,remaining:obligationSummary(item).remaining})).filter(x=>x.remaining>=.005);const remaining=roundMoney(pending.reduce((s,x)=>s+x.remaining,0)),liquidity=roundMoney(acs().reduce((s,[account])=>s+balanceAtDate(account,period.asOf),0));return {period,actual,income,expense,pending,remaining,liquidity,available:roundMoney(liquidity-remaining)};}
